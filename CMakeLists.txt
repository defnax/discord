cmake_minimum_required(VERSION 3.22)
project(DiscordBridgePlugin LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt5 5.15 REQUIRED COMPONENTS Core Widgets Network)
get_filename_component(RETROSHARE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(RETROSHARE_ROOT "${RETROSHARE_ROOT}" CACHE PATH "RetroShare source tree")
add_library(Discord MODULE DiscordPlugin.cpp DiscordPlugin.h DiscordBridge.cpp DiscordBridge.h BridgeMessages.h Discord_images.qrc)
set_target_properties(Discord PROPERTIES PREFIX "")
target_include_directories(Discord PRIVATE
    "${RETROSHARE_ROOT}/libretroshare/src"
    "${RETROSHARE_ROOT}/supportlibs/rapidjson/include")
target_link_libraries(Discord PRIVATE Qt5::Core Qt5::Widgets Qt5::Network)
target_compile_definitions(Discord PRIVATE RS_NO_WARN_DEPRECATED)
if(WIN32)
    target_compile_definitions(Discord PRIVATE WINDOWS_SYS WIN32_LEAN_AND_MEAN)
    if(TARGET retroshare)
        target_link_libraries(Discord PRIVATE retroshare)
    else()
        find_library(RETROSHARE_LIBRARY NAMES retroshare
            HINTS "${RETROSHARE_ROOT}/libretroshare/src/lib" REQUIRED)
        target_link_libraries(Discord PRIVATE "${RETROSHARE_LIBRARY}")
    endif()
endif()
install(TARGETS Discord LIBRARY DESTINATION lib/retroshare/plugins RUNTIME DESTINATION lib/retroshare/plugins)
option(DISCORD_BUILD_TESTS "Build offline bridge tests" ON)
if(DISCORD_BUILD_TESTS)
    enable_testing()
    find_package(Qt5 5.15 REQUIRED COMPONENTS Test)
    add_executable(BridgeMessagesTest tests/BridgeMessagesTest.cpp)
    target_include_directories(BridgeMessagesTest PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(BridgeMessagesTest PRIVATE Qt5::Test Qt5::Widgets)
    add_test(NAME BridgeMessages COMMAND BridgeMessagesTest -platform offscreen)
    add_executable(DiscordBridgeTest tests/DiscordBridgeTest.cpp DiscordBridge.cpp DiscordBridge.h)
    target_include_directories(DiscordBridgeTest PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}"
        "${RETROSHARE_ROOT}/libretroshare/src" "${RETROSHARE_ROOT}/supportlibs/rapidjson/include")
    target_link_libraries(DiscordBridgeTest PRIVATE Qt5::Test Qt5::Widgets Qt5::Network)
    if(TARGET retroshare)
        target_link_libraries(DiscordBridgeTest PRIVATE retroshare)
    else()
        find_library(RETROSHARE_LIBRARY NAMES retroshare
            HINTS "${RETROSHARE_ROOT}/libretroshare/src/lib" REQUIRED)
        target_link_libraries(DiscordBridgeTest PRIVATE "${RETROSHARE_LIBRARY}")
    endif()
    if(WIN32)
        target_compile_definitions(DiscordBridgeTest PRIVATE WINDOWS_SYS WIN32_LEAN_AND_MEAN)
    endif()
    add_test(NAME DiscordBridge COMMAND DiscordBridgeTest -platform offscreen)
    if(WIN32)
        set(RETROSHARE_RUNTIME_DIRS
            "${RETROSHARE_ROOT}/libretroshare/src/lib;${RETROSHARE_ROOT}/supportlibs/librnp/Build/src/lib"
            CACHE STRING "Additional DLL directories for offline tests")
        foreach(runtime_dir IN LISTS RETROSHARE_RUNTIME_DIRS)
            set_property(TEST DiscordBridge APPEND PROPERTY ENVIRONMENT_MODIFICATION
                "PATH=path_list_prepend:${runtime_dir}")
        endforeach()
    endif()
endif()
